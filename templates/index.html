import os import random from pathlib import Path from PIL import Image, ImageDraw from moviepy.editor import AudioFileClip, ImageClip, concatenate_videoclips, CompositeVideoClip, TextClip class VideoComposer: def __init__(self): self.output_dir = Path('videos') self.slide_dir = Path('temp_frames') self.output_dir.mkdir(exist_ok=True, parents=True) self.slide_dir.mkdir(exist_ok=True, parents=True) def create_visuals(self, topic: str, script: str): """Create slide visual assets from the narration script.""" paragraphs = [p.strip() for p in script.split('\n\n') if p.strip()] slides = [] for index, paragraph in enumerate(paragraphs[:12]): title = f"{topic[:30]}" if index == 0 else f"Section {index}" slide_path = self._create_slide(title, paragraph, index) slides.append(slide_path) if not slides: slides.append(self._create_slide(topic, 'Overview', 0)) return slides def _create_slide(self, title: str, text: str, index: int) -> str: width, height = 1920, 1080 image = Image.new('RGB', (width, height), color=(25, 30, 50)) draw = ImageDraw.Draw(image) # Background accent shapes for i in range(0, 1600, 250): draw.rounded_rectangle( [i, 100 + (index % 3) * 50, i + 150, 260 + (index % 3) * 50], radius=20, fill=(random.randint(40, 90), random.randint(70, 130), random.randint(110, 180)) ) draw.text((100, 100), title, fill=(255, 255, 255), font=None) self._draw_wrapped_text(draw, text, (100, 260), 82, max_width=1700) slide_path = self.slide_dir / f'slide_{index}.png' image.save(slide_path) return str(slide_path) def _draw_wrapped_text(self, draw, text: str, start_xy, font_size: int, max_width: int = 1700): words = text.split() lines = [] current = "" for word in words: test = f"{current} {word}".strip() if len(test) <= 80: current = test else: lines.append(current) current = word if current: lines.append(current) x, y = start_xy for line in lines[:10]: draw.text((x, y), line, fill=(240, 240, 255)) y += font_size + 8 def compose_video(self, audio_path: str, slides, title: str) -> str: audio = AudioFileClip(audio_path) clips = [] duration_per_slide = max(4, audio.duration / max(len(slides), 1)) for i, slide in enumerate(slides): image_clip = ImageClip(slide).set_duration(duration_per_slide) title_clip = TextClip( text=f"{title[:40]}", fontsize=30, color='white', font='DejaVu-Sans', method='label' ).set_position(('center', 20)).set_duration(duration_per_slide) clip = CompositeVideoClip([image_clip, title_clip]) clip = clip.set_duration(duration_per_slide) clips.append(clip) video = concatenate_videoclips(clips, method='compose') video = video.set_audio(audio) output_path = self.output_dir / f'{title.replace(" ", "_")}_final.mp4' video.write_videofile(str(output_path), fps=24, codec='libx264', audio_codec='aac', logger=None, verbose=False) return str(output_path)
